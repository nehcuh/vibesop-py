    sys.path.pop(0)
import json, tempfile

from unittest.mock import patch
import numpy as np
from vibesop.core.instinct.learner import InstinctLearner
from vibesop.core.observability.models import Span
from vibesop.core.observability.span_writer import SpanWriter

with tempfile.TemporaryDirectory() as temp:
    root = Path(temp)
    path = root/'instincts.jsonl'
    a = InstinctLearner(path)
    inst = a.learn('debug flaky tests','builtin/systematic-debugging')
    b = InstinctLearner(path)
    a.record_outcome(inst.id, True)
    b.record_outcome(inst.id, True)
    after = InstinctLearner(path).get_instinct_for_query('debug flaky tests')
    print('COUNTER_LOSS',json.dumps({'expected_success_count':2,'actual_success_count':after.success_count}))
    writer = SpanWriter(root/'spans.jsonl')
    for i in range(3):
        span = Span(id=f's{i}', trace_id=f'trace{i}', name='route:query', span_kind='task', task_id='same-task', project_id='test', input_data={'query':'debug flaky tests'}, output_data={'skill_id':'builtin/systematic-debugging'}, metadata={'skill_id':'builtin/systematic-debugging'})
        span.set_ok()
        writer.write_span(span)
    cache = EmbeddingCache(root/'embed.npz')
    with patch.object(cache,'_compute',return_value=np.ones(384,dtype=np.float32)):
        result = recall_similar('debug flaky tests', writer.query_recent(),cache=cache,learner=InstinctLearner(path))[0]
    print('DISK_RECALL_SKILL',json.dumps({'expected_skill':'builtin/systematic-debugging','actual_skill':result.skill_id,'is_gold':result.is_gold}))
    corrupted = root/'bad.jsonl'
    corrupted.write_text(json.dumps({'id':'x','pattern':'x','action':'x','created_at':'bad-date'})+'\n')
    try:
        InstinctLearner(corrupted)
        print('BAD_DATE load_ok')
    except Exception as exc:
        print('BAD_DATE',type(exc).__name__,str(exc))
